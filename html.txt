        // TAB NAVIGATION SYSTEM
        function switchTab(tabId) {
            document.querySelectorAll('.tab-view').forEach(el => el.classList.add('hidden'));
            document.getElementById('tab-' + tabId).classList.remove('hidden');

            document.querySelectorAll('.nav-btn').forEach(btn => {
                btn.className = "nav-btn px-5 py-2.5 rounded-xl text-sm font-bold transition-all flex items-center space-x-2.5 text-slate-400 hover:text-white hover:bg-white/5";
            });
            const activeNav = document.getElementById('nav-' + tabId);
            if(activeNav) {
                activeNav.className = "nav-btn px-5 py-2.5 rounded-xl text-sm font-bold transition-all flex items-center space-x-2.5 bg-gradient-to-r from-violet-600 to-pink-600 text-white shadow-neon-purple";
            }

            document.querySelectorAll('.mob-btn').forEach(btn => {
                btn.className = "mob-btn flex flex-col items-center text-slate-400 space-y-1";
            });
            const activeMob = document.getElementById('mob-' + tabId);
            if(activeMob) {
                activeMob.className = "mob-btn flex flex-col items-center text-violet-400 space-y-1";
            }

            window.scrollTo({ top: 0, behavior: 'smooth' });
        }

            const code = currentActiveSquad ? currentActiveSquad.pinCode : "482-910";

        function filterNotes() {
            const q = document.getElementById('noteSearch').value.toLowerCase();
            const subject = document.getElementById('noteSubjectFilter').value;
            const year = document.getElementById('noteYearFilter').value;

            const filtered = notesData.filter(n => {
                const matchQ = n.title.toLowerCase().includes(q) || n.desc.toLowerCase().includes(q);
                const matchSub = subject === 'all' || n.subject === subject;
                const matchYear = year === 'all' || n.year === year;
                return matchQ && matchSub && matchYear;
            });
            renderNotes(filtered);
        }

        // LO-FI MUSIC PLAYER
        function toggleLofiPlayer() {
            const audio = document.getElementById('lofiAudio');
            const icon = document.getElementById('lofiIcon');
            if(isLofiPlaying) {
                audio.pause();
                icon.className = "fa-solid fa-play text-xs";
                isLofiPlaying = false;
            } else {
                audio.play().then(() => {
                    icon.className = "fa-solid fa-pause text-xs";
                    isLofiPlaying = true;
                }).catch(() => {
                    showToast("Clicca per attivare l'audio lo-fi di sottofondo!");
                });
            }
        }

        // TOAST NOTIFICATIONS
        function showToast(msg) {
            const toast = document.createElement('div');
            toast.className = "fixed bottom-8 right-8 z-50 bg-slate-900 border border-violet-500/40 text-white px-5 py-3.5 rounded-2xl shadow-neon-purple flex items-center space-x-3 text-xs font-bold";
            toast.innerHTML = `<i class="fa-solid fa-circle-info text-cyan-400 text-base"></i><span>${msg}</span>`;
            document.body.appendChild(toast);
            setTimeout(() => {
                toast.style.opacity = '0';
                toast.style.transition = 'opacity 0.3s ease';
                setTimeout(() => toast.remove(), 300);
            }, 3000);
        }
